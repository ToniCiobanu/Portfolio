🗄️ SQL Database Schema & Queries

-- ============================================================ -- AUTOMOTIVE REPAIR COST ANALYSIS DATABASE -- Author: Toni Ciobanu | Portfolio Project -- ============================================================ CREATE TABLE vehicles ( vehicle_id INT PRIMARY KEY, make VARCHAR(50) NOT NULL, model VARCHAR(50) NOT NULL, year INT NOT NULL, odometer_km INT, acquisition_date DATE ); CREATE TABLE repair_categories ( category_id INT PRIMARY KEY, category_name VARCHAR(100) NOT NULL, category_type VARCHAR(50) ); CREATE TABLE repair_line_items ( item_id INT PRIMARY KEY, vehicle_id INT REFERENCES vehicles(vehicle_id), category_id INT REFERENCES repair_categories(category_id), description VARCHAR(200) NOT NULL, diy_parts_cost DECIMAL(10,2) NOT NULL, mechanic_est_total DECIMAL(10,2), -- Verified quote savings DECIMAL(10,2), -- mechanic_est_total - diy_parts_cost savings_pct DECIMAL(5,2), -- savings / mechanic_est_total * 100 parts_source VARCHAR(50), diy_performed BOOLEAN DEFAULT TRUE );

Query 1: Cost Summary by Vehicle

-- Total DIY vs Mechanic Cost by Vehicle SELECT v.make || ' ' || v.model || ' ' || v.year AS vehicle, v.odometer_km, COUNT(r.item_id) AS num_repairs, ROUND(SUM(r.diy_parts_cost), 2) AS total_diy_cost, ROUND(SUM(r.mechanic_est_total), 2) AS total_mechanic_est, ROUND(SUM(r.savings), 2) AS total_savings, ROUND(SUM(r.savings) / SUM(r.mechanic_est_total) * 100, 2) AS savings_pct FROM vehicles v JOIN repair_line_items r ON v.vehicle_id = r.vehicle_id GROUP BY v.vehicle_id, v.make, v.model, v.year, v.odometer_km ORDER BY total_savings DESC;

Query 2: Best & Worst Value Repairs (Window Function)

-- Rank all repairs by savings % using RANK() window function SELECT v.make || ' ' || v.model AS vehicle, r.description, rc.category_name, r.diy_parts_cost, r.mechanic_est_total, r.savings, r.savings_pct, RANK() OVER (ORDER BY r.savings_pct DESC) AS best_value_rank, RANK() OVER (ORDER BY r.savings_pct ASC) AS worst_value_rank FROM repair_line_items r JOIN vehicles v ON r.vehicle_id = v.vehicle_id JOIN repair_categories rc ON r.category_id = rc.category_id ORDER BY best_value_rank;

💡 Key Insights & Findings

1. 85.93% Savings Rate Across Both Vehicles

Across 23 repairs, DIY methods saved $10,089.26 (85.93%) vs. verified mechanic/dealership quotes. Both vehicles achieved nearly identical rates — Pontiac 85.50%, Elantra 86.36% — suggesting the savings model is consistent regardless of vehicle age.

2. Junkyard Sourcing = Highest ROI at 1,234%

The two junkyard-sourced items (hood $100, brake light housing $5) generated $1,295.48 in savings — a 1,234% ROI. The hood alone saved $1,242.68 vs. dealership pricing of $1,342.68. For structural non-wear parts, junkyard sourcing is unmatched.

3. Safety Inspection Items: Highest Category Savings Rate (93.51%)

5 safety inspection failure items cost only $192 DIY vs. $2,956.43 at a mechanic — a 93.51% savings rate. The front bumper repair ($30 DIY vs. $1,075.76 mechanic) is the single most striking example of DIY value in this dataset.

4. Best Value: Sway Bar Link (97.4% savings)

A $6.50 sway bar link from RockAuto would cost $250 at a mechanic — a 3,746% ROI and the highest savings percentage in the dataset. Small suspension components like this represent the highest-leverage DIY repairs.

5. Worst Value: Battery Replacement (26.55% savings)

The battery was purchased at retail ($220.34) while the mechanic quote was only $300 — a narrow $79.66 spread. Mechanic markup on batteries is low because they source them at wholesale. Batteries are the weakest case for DIY in this dataset.
